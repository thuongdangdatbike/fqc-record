Cho phép trình duyệt sử dụng camera khi được hỏi.

let scanner = null; let scannedText = ''; function setMessage(text) { document.getElementById('message').textContent = text; } async function startScanner() { if (!window.isSecureContext) { setMessage('Trang chưa ở ngữ cảnh HTTPS an toàn. Hãy kiểm tra URL GitHub Pages.'); return; } if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { setMessage('Trình duyệt không hỗ trợ camera. Hãy mở trang bằng Chrome hoặc Safari.'); return; } try { if (!scanner) scanner = new Html5Qrcode('reader'); document.getElementById('startBtn').disabled = true; setMessage('Đang mở camera...'); await scanner.start( { facingMode: 'environment' }, { fps: 10, qrbox: { width: 250, height: 250 } }, async function(decodedText) { scannedText = decodedText; document.getElementById('result').textContent = decodedText; document.getElementById('continueBtn').disabled = false; setMessage('Đọc QR thành công. Nhấn Tiếp tục để sang FQC record.'); try { await scanner.stop(); } catch (e) {} document.getElementById('startBtn').disabled = false; }, function() {} ); setMessage('Đưa mã QR vào trong khung camera.'); } catch (error) { document.getElementById('startBtn').disabled = false; setMessage( 'Không mở được camera: ' + (error.message || error) + '. Hãy kiểm tra quyền camera hoặc nhập mã thủ công.' ); } } function openManual() { const value = prompt('Nhập chính xác nội dung mã QR:'); if (value && value.trim()) { scannedText = value.trim(); document.getElementById('result').textContent = scannedText; document.getElementById('continueBtn').disabled = false; setMessage('Đã nhập mã. Nhấn Tiếp tục để sang FQC record.'); } } function continueToFQC() { if (!scannedText) { setMessage('Bạn chưa quét hoặc nhập mã QR.'); return; } if (!FQC_APP_URL.startsWith('https://') || !FQC_APP_URL.includes('/exec')) { setMessage('Bạn chưa điền đúng URL web app FQC record trong code.'); return; } const target = FQC_APP_URL + '?qr=' + encodeURIComponent(scannedText); window.location.href = target; }
